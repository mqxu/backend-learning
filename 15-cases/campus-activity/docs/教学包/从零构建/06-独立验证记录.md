```bash
# 1. 克隆（约 1 秒）
rm -rf /tmp/verify-tutorial
git clone -b tutorial/from-zero \
  docs/教学包/从零构建/campus-activity-tutorial.bundle /tmp/verify-tutorial

# 2. C1 / C2（只验编译；可选起后端看 401 → 200）
git checkout ba086b2 && (cd server && mvn -o -DskipTests compile)
git checkout ee173f6 && (cd server && mvn -o -DskipTests compile)

# 3. C3（独立库覆盖，避免污染校区库）
mysql -uroot -p123456 -e "CREATE DATABASE IF NOT EXISTS campus_activity_verify DEFAULT CHARSET utf8mb4"
git checkout 5a5c97b
(cd server/src/main/resources/db && \
  mysql -h127.0.0.1 -P3306 -uroot -p123456 campus_activity_verify < schema.sql && \
  mysql -h127.0.0.1 -P3306 -uroot -p123456 campus_activity_verify < data.sql)
(cd server && mvn -o test \
  -Dspring.datasource.url="jdbc:mysql://127.0.0.1:3306/campus_activity_verify?useUnicode=true&characterEncoding=utf8&serverTimezone=Asia/Shanghai&useSSL=false&allowPublicKeyRetrieval=true")

# 4. C4 / C5（node_modules 可软链主项目；否则先 pnpm install）
git checkout 067c5b3 && (cd app   && ln -sfn <主项目>/app/node_modules   node_modules && pnpm type-check && pnpm build:h5)
git checkout 5f4da02 && (cd admin && ln -sfn <主项目>/admin/node_modules node_modules && pnpm type-check && pnpm build)

# 5. C6（先起后端再冒烟）
git checkout 4e7d08a6
(cd server && mvn -o spring-boot:run \
  "-Dspring-boot.run.arguments=--spring.datasource.url=<上面的 verify URL>") &
until curl -s http://127.0.0.1:8016/api/ping | grep -q '"code":0'; do sleep 1; done
./scripts/smoke.sh            # 全部通过：33 项

# 6. 清理
kill "$(lsof -nP -iTCP:8016 -sTCP:LISTEN -t)" 2>/dev/null
mysql -uroot -p123456 -e "DROP DATABASE IF EXISTS campus_activity_verify"
for k in $(redis-cli -a abc123 --no-auth-warning -n 11 keys 'campus:*'); do
  redis-cli -a abc123 --no-auth-warning -n 11 del "$k" >/dev/null; done
```
