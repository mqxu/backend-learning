### 1.3 本教程的拆解方式（重要读法）

9 个提交按文件拆出：**每个文件首次出现的提交里就是最终版本**，真正逐步演化的只有 `router/index.ts`（D03 建立、D04–D09 各加一条）和 `main.ts`/`App.vue`（D01 占位 → D03 真实入口）。因此会出现：

1. D04 的布局里已有 D09 才用的“素材上传”菜单，D02 的 `api/index.ts` 已写好 `uploadFile()`（到 D09 才被页面调用）；D04–D08 侧边栏有 `/file` 入口但路由 D09 才注册，点击会被兜底带回工作台；

这些都是拆解产物，不是路由错误。

**读法**：以“文件首次出现”为锚点；每节的“路由变化”单独看，正好能看清路由表如何一行行长大。

**关键代码**：`api/request.ts` 是后台“请求总线”，D03 登录、D04 退出、D09 上传都走它（完整 102 行见 `git show af54636:admin/src/api/request.ts`）：

```ts
import axios, { type AxiosError, type AxiosInstance, type AxiosRequestConfig, type AxiosResponse, type InternalAxiosRequestConfig } from 'axios'
import { ElMessage } from 'element-plus'
import type { ApiResult, UserInfo } from '@/types/api'

const TOKEN_KEY = 'campus_admin_token'
const USER_KEY = 'campus_admin_user'

export function getToken(): string {
  return localStorage.getItem(TOKEN_KEY) || ''
}

export function setAuth(token: string, user?: UserInfo | null): void {
  localStorage.setItem(TOKEN_KEY, token)
  localStorage.setItem(USER_KEY, JSON.stringify(user || {}))
}

export function clearAuth(): void {
  localStorage.removeItem(TOKEN_KEY)
  localStorage.removeItem(USER_KEY)
}

function redirectToLogin(): void {
  clearAuth()
  if (!window.location.hash.startsWith('#/login')) window.location.hash = '#/login'
}

function unwrapResponse(response: AxiosResponse): unknown {
  const body: unknown = response.data
  if (body && typeof body === 'object' && 'code' in body) {
    const result = body as ApiResult<unknown>
    if (result.code === 0) return result.data
    const message = result.message || '请求失败'
    ElMessage.error(message)
    throw new Error(message)
  }
  return body
}

const service: AxiosInstance = axios.create({ baseURL: '/api', timeout: 15000 })

service.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const token = getToken()
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

service.interceptors.response.use(
  (response: AxiosResponse) => unwrapResponse(response) as unknown as AxiosResponse,
  (error: AxiosError<ApiResult<unknown>>) => {
    const status = error.response?.status
    if (status === 401) {
      ElMessage.error('登录状态已失效，请重新登录')
      redirectToLogin()
      return Promise.reject(new Error('未登录或登录已过期'))
    }
    if (status === 403) {
      ElMessage.error('没有权限执行该操作')
      return Promise.reject(new Error('没有权限执行该操作'))
    }
    const message = error.response?.data?.message || error.message || '网络异常，请稍后重试'
    ElMessage.error(message)
    return Promise.reject(new Error(message))
  }
)

export interface HttpClient {
  get<T = unknown>(url: string, config?: AxiosRequestConfig): Promise<T>
  post<T = unknown>(url: string, data?: unknown, config?: AxiosRequestConfig): Promise<T>
}

export default service as unknown as HttpClient
```
